\begin{frame}{Research Questions and Hypotheses}
  \begin{table}
    \centering
    \begin{tabularx}{\textwidth}{@{}l >{\raggedright\arraybackslash}X@{}}
      \toprule
      Research question & Key prediction \\
      \midrule
      \textbf{RQ1 Rise:} what does a fashion displace? &
        More foundation-model (FM) papers in a topic $\to$ fewer new-architecture papers \rc{H1d} \\
      \addlinespace
      \textbf{RQ2 Following:} who introduces new work? &
        Newcomers imitate: fewer new-method papers, more benchmark papers \rc{H2b, H2c} \\
      \addlinespace
      \textbf{RQ3 Reward:} are followers rewarded? &
        Crowded topics $\to$ lower acceptance \rc{H3a} \\
      \bottomrule
    \end{tabularx}
  \end{table}
  \gnote{\small All predictions were written down before any test was run (10 hypotheses in total).}
\end{frame}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Who drives new work? Two views}
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item \emph{Fashion and herding:} newcomers imitate\\
          \rc{Abrahamson, 1991; Bikhchandani et al., 1992}
        \item \emph{Outsider innovation:} newcomers bring new ideas\\
          \rc{Foster et al., 2015; Azoulay et al., 2019}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{What we add to meta-research on machine learning}\\
      \rc{Koch et al., 2021; Birhane et al., 2022}
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item Content trends across all submissions, not only accepted papers
        \item Topics defined without hindsight
        \item Tests registered in advance \rc{Nosek et al., 2018}, plus a held-out replication
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

        \caption{Topic diversity per year under three topic maps.}
      \end{figure}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Topic map}
        \small The 100 topic clusters, fitted on papers from chosen years; every paper is then assigned to its
        nearest topic.
      \end{block}
      \vspace{0.5em}
      Each map finds peak diversity in the years it was fitted on.
      \vspace{0.8em}

      $\Rightarrow$ \alert{The fitting window shapes the history one sees.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Consequence: Conclusions Depend on the Map}
  \begin{itemize}
    \item Emerging topics (${\geq}3\times$ growth): \textbf{8} on the 2017--20 map vs.\ \textbf{29} on the
      all-years map.
    \item Do topics age, from new methods towards benchmarks? All-years map: no. 2017--20 map: yes.
  \end{itemize}
  \vspace{-0.4em}
  \begin{figure}
    \includegraphics[width=0.62\textwidth]{figs/lifecycle_flip.pdf}
    \caption{Same papers, same model: coefficient on topic maturity, 95\% CI (2017--20 rows exploratory).}
  \end{figure}
  \vspace{-0.6em}
  \centering $\Rightarrow$ Every registered result must also hold on the 2017--20 map and a year-balanced map.
\end{frame}

\section{Empirical Strategy}

\begin{frame}{Empirical Strategy}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Model} (linear probability)
      \[
        y_i \;=\; \beta\, x_i \;+\; \mu_{\text{topic}(i)} \;+\; \tau_{\text{year}(i)} \;+\; \varepsilon_i
      \]
      \small
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item $y_i$: paper type or acceptance (0/1); $x_i$: e.g.\ newcomer paper
        \item Papers compared within the same topic and year; standard errors clustered by topic
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Supported only if}
        \small
        \begin{enumerate}
          \item predicted sign, Holm-adjusted $p < 0.05$
          \item same sign in ${\geq}75\%$ of alternative specifications
          \item same sign on both hindsight checks
        \end{enumerate}
      \end{block}
      \begin{block}{Protocol}
        \small Registered in advance, acceptance locked, code frozen, run once; held-out tests registered before
        their run.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

  \begin{columns}[c]

        \caption{Discovery; within topic and year; grey: predicted side.}
      \end{figure}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{table}
        \centering\small
        \begin{tabular}{@{}lrr@{}}
          \toprule
          Newcomer paper (pp) & Method & Benchmark \\
          \midrule
          Predicted & $-$ & $+$ \\
          Discovery & $+3.7$ & $-1.7$ \\
          Held-out & $+3.6$ & $-2.1$ \\

      \end{table}
      \small

      \begin{block}{Reading}
        \small Consistent with outsider innovation, not with herding.
      \end{block}

  \end{columns}

\begin{frame}{RQ3: Is Following the Fashion Rewarded?}
  \begin{table}
    \centering\small
    \begin{tabularx}{\textwidth}{@{}X r r@{}}
      \toprule
      Association with acceptance (pp, 95\% CI) & Discovery & Held-out \\

\section{Conclusion}

\begin{frame}{Contributions}
  \begin{enumerate}\setlength{\itemsep}{0.9em}
    \item \textbf{Method:} topic maps fitted with hindsight can manufacture trends; an early-year map is a cheap
      check.
    \item \textbf{Design:} registered, outcome-blind, single-run protocol, with a held-out replication.
    \item \textbf{Findings} (associations):
      \begin{itemize}
        \item where foundation models spread, new architectures fade \alert{(replicated)}
        \item newcomers lean towards new methods, not imitation \alert{(replicated)}
        \item no detectable acceptance reward for following the fashion
      \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Limitations and Future Work}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Limitations}
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item One venue; associations, not causes
        \item LLM labels: consistent ($\kappa = 0.93$), not yet human-validated
        \item Held-out papers: a released former split, not certified untouched
        \item 2021 author-ID change undercounts incumbents
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Future work}
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item A prospective test on future ICLR submissions
        \item ACL Anthology / ARR
        \item Human validation of contribution labels
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
