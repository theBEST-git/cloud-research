\begin{frame}{Related Work and Positioning}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Two views of who drives new work}
      \begin{itemize}
        \item \emph{Fashion and herding}: ideas spread by imitation \rc{Abrahamson, 1991; Bikhchandani et al., 1992}
          $\Rightarrow$ newcomers follow
        \item \emph{Outsider innovation}: new entrants bring new ideas \rc{Azoulay et al., 2019};
          tradition vs.\ innovation \rc{Foster et al., 2015}
        \item Team size: large teams develop, small teams disrupt \rc{Wu et al., 2019}
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Meta-research on ML}
      \begin{itemize}
        \item Benchmark concentration \rc{Koch et al., 2021}
        \item Values encoded in ML papers \rc{Birhane et al., 2022}
      \end{itemize}
      \vspace{0.3em}
      \textbf{Measurement challenges we address}
      \begin{enumerate}
        \item Survivorship: only published papers observed
        \item Hindsight: topics defined on the full period
        \item Forking paths \rc{Gelman \& Loken, 2014} $\Rightarrow$ pre-registration \rc{Nosek et al., 2018}
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\section{Empirical Strategy}

\begin{frame}{Empirical Strategy}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Specification} (linear probability model)
      \[
        y_i \;=\; \beta\, x_i \;+\; \mu_{k(i)} \;+\; \tau_{t(i)} \;+\; \varepsilon_i
      \]
      \vspace{-1.2em}
      \footnotesize
      \begin{itemize}
        \item $y_i \in \{0,1\}$: paper type or acceptance; $x_i$: e.g.\ newcomer paper
        \item $\mu_k$, $\tau_t$: topic and year fixed effects (topic $\times$ year for RQ2)
        \item Standard errors clustered by topic (100 clusters)
      \end{itemize}
      \small\textbf{Decision rule} (registered). Supported only if:
      \footnotesize
      \begin{enumerate}
        \item predicted sign, and Holm-adjusted $p < 0.05$;
        \item same sign in ${\geq}75\%$ of 9--11 alternative specifications;
        \item same sign under both hindsight checks.
      \end{enumerate}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Protocol}
        \footnotesize
        Pre-registered; variables built blind to acceptance; pipeline tested on 12,400 simulation runs; code and
        inputs frozen (SHA-256); a single run; every estimate reported.
      \end{block}
      \begin{block}{Simulation finding}
        \footnotesize Under realistic no-effect data, clustered SEs rejected up to \alert{12.6\%} at nominal 5\%.
        The RQ2 results therefore get two extra checks: two-way clustering and a stress test.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Contributions}
  \begin{enumerate}
    \item \textbf{Methodological.} Topic maps fitted with hindsight manufacture emerging topics and can reverse trend
      conclusions; early-year maps provide a cheap check.
    \item \textbf{Design.} A pre-registered, outcome-blind, simulation-checked, single-run protocol for
      science-of-science studies.
    \item \textbf{Substantive.}
      \begin{itemize}
        \item Where foundation models spread, new architectures fade (H1d, supported).
        \item Newcomer papers lean towards new methods and away from benchmarks, against the herding prediction
          (H2b/H2c).
        \item Following fashions is not rewarded; areas such as federated learning fall out of favour as the
          field moves on, not because of who writes the papers (post hoc).
      \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Limitations and Future Work}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Limitations}
      \small
      \begin{itemize}
        \item One venue (ICLR)
        \item Associations, not causal effects
        \item LLM labels: consistent ($\kappa = 0.93$), not yet human-validated
        \item 2021 author-ID change undercounts incumbents (expected bias towards zero)
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Future work}
      \small
      \begin{itemize}
        \item Replication on the held-out ICLR split (five hypotheses registered, incl.\ federated learning)
        \item Extension to ACL Anthology / ARR
        \item Human validation of contribution labels
        \item Review scores: reviewers vs.\ final decisions
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{A3. Calibration of Standard Errors}
  \small
  \begin{itemize}
    \item Before seeing any result: 12,400 simulation runs, including no-effect scenarios with realistic topic
      structure.
    \item When true topic structure cuts across the 100 topics, the standard test rejected \alert{12.6\%} (H2b) and
      10.0\% (H2c) of the time at nominal 5\%. The wild bootstrap did not fix this.
    \item Added before the run:
      \begin{itemize}
        \item two-way clustering (two topic maps at once);
        \item a stress test that deflates each $t$-statistic by the simulated inflation (1.31 / 1.22).
      \end{itemize}
    \item RQ2 results: $t = 6.7$ and $-4.4$. After the stress test, Holm $p$ = $4\times10^{-6}$ and $1\times10^{-3}$.
  \end{itemize}
  \begin{block}{Lesson}
    Check the test on data shaped like yours before trusting its $p$-values.
  \end{block}
\end{frame}

\begin{frame}{A11. Definitions}
  \small
  \begin{description}[Benchmark \& eval.]
    \item[Topic] one of 100 clusters of paper embeddings (title + abstract)
    \item[Topic maturity] $\log(1 + \text{earlier papers in the topic})$; $+1$ = 2.7$\times$ more papers
    \item[Hot topic] top fifth of topics by last year's growth
    \item[Newcomer paper] no author has an earlier ICLR submission (any outcome) in that topic
    \item[FM adoption] share of the topic's papers last year that built on a foundation model
    \item[Benchmark \& eval.] primary contribution is a dataset, benchmark or evaluation
    \item[Model] linear probability model with topic and year fixed effects (topic $\times$ year cells for RQ2);
      errors clustered by topic
    \item[pp] percentage points
  \end{description}
\end{frame}
